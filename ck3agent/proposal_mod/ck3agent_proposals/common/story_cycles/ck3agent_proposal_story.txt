# ck3agent incoming-proposal queue: one story per held offer, owned by the
# seated recipient (lane QMOD; docs/design/incoming-proposals.md, "Timing").
# Nothing in the engine limits a story's life: this cycle ages it in 10-day
# steps, voids it when the offer can no longer be answered, applies the class's
# lapse default after ck3agent_proposal_lapse_steps steps (400 days; by=lapse),
# and ends a resolved or void story on its next step.
ck3agent_proposal = {
	on_setup = { }

	on_end = { }

	# The seat died: every open offer is void, its sender released.
	on_owner_death = {
		if = {
			limit = { ck3agent_proposal_open_trigger = yes }
			ck3agent_proposal_restore_scopes_effect = yes
			save_scope_value_as = { name = ck3a_void value = 4 }
			ck3agent_proposal_void_effect = yes
		}
		end_story = yes
	}

	effect_group = {
		days = 10
		first_valid = {
			triggered_effect = {
				trigger = { NOT = { ck3agent_proposal_open_trigger = yes } }
				effect = { end_story = yes }
			}
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					change_variable = { name = ck3agent_proposal_age add = 1 }
					ck3agent_proposal_restore_scopes_effect = yes
					ck3agent_proposal_check_void_effect = yes
					if = {
						limit = { scope:ck3a_void > 0 }
						ck3agent_proposal_void_effect = yes
					}
					else_if = {
						limit = { var:ck3agent_proposal_age >= ck3agent_proposal_lapse_steps }
						ck3agent_proposal_lapse_effect = yes
					}
				}
			}
		}
	}
}
